On the BMP585, **SDO picks the I²C address**. It isn't a data line here.

- SDO → GND gives address **0x46**
- SDO → 3V3 gives address **0x47**

The chip has to see one or the other, so SDO can't be left floating. If it floats, the address can change from one power-up to the next and the Teensy may not find the sensor.

**Why 3V3 and not GND:** your flight firmware ([Config.h:245](C:/Users/kalbh/Downloads/vritra-gcs-master/vritra-gcs-master/firmware/vritra_fc/Config.h:245)) has a comment saying 0x47 showed up in your own I²C scan and 0x46 didn't. **On Board 2 it's mainly for consistency.** The other chips on that bus are the LSM6DSO32 (0x6A) and the HDC3020 (0x44), so neither choice would clash. GND would work too, because the firmware tries 0x47 first, then 0x46, and checks the chip ID either way. I kept 3V3 so every BMP585 uses the same wiring and the same address.

Yes, sharing the GNSS between the Teensys is fine. The L89 talks over a serial line (UART), which has no addresses. It continuously sends its NMEA position sentences out of one TX pin, and every Teensy wired to that pin simply listens. The boards you're printing are wired exactly that way:

|---|---|---|
| Board 1 MAIN | pin 7 (Serial2 RX) | straight from the L89 TX pin |
| Board 2 BACKUP | pin 0 (Serial1 RX) | L89 TX → 1 kΩ R6 → MAIN J5 pin 2 → link cable → BACKUP J2 pin 2 |
| Board 4 PARACHUTE | pin 0 (Serial1 RX) | L89 TX → 1 kΩ R5 → MAIN J8 pin 1 → cable → PARACHUTE J5 pin 1 |

Only MAIN can talk back to the GNSS (pin 8 → L89 RX); the other two only listen. The 1 kΩ resistors protect the MAIN GNSS line if a cable shorts or the other Teensy is switched off.

```cpp
/*
 * VRITRA Board 4 (PARACHUTE, Teensy 4.0) — bench test
 *   - I2C scan of Wire (pins 18 SDA / 19 SCL)
 *   - LSM6DSO32 @0x6A: raw accel / gyro / temperature + scaled values
 *   - BMP585    @0x47: raw temperature / pressure + °C / Pa / altitude
 *   - one servo sweeping continuously (default J12 = MG90S #1, Teensy pin 4)
 * Serial monitor (any baud). Keys: 's' = stop/start servo, 'c' = centre servo,
 *                                  '2'..'5' / 'x' = choose servo pin 2,3,4,5 / 10 (see table below).
 * Only Teensyduino built-in libraries are used (Wire, Servo).
 *
 * Board 4 servo plugs (− + S):  J10 STEER L = pin 2 (8 V)   J11 STEER R = pin 3 (8 V)
 *                               J12 BAY     = pin 4 (5 V)   J13 PETRI   = pin 5 (5 V)   J16 EXTRA = pin 10 (5 V)
 */
#include <Wire.h>
#include <Servo.h>

// ---------------- pins / addresses (Board 4 design) ----------------
static uint8_t servoPin = 4;           // J12 MG90S #1 (5 V servo rail from J6)
const uint8_t LSM_ADDR  = 0x6A;        // LSM6DSO32, DO -> GND
const uint8_t BMP_ADDR  = 0x47;        // BMP585, SDO -> 3V3

// servo sweep (MG90S safe range 1000..2000 us; widen slowly if you need more travel)
const int SERVO_MIN_US = 1000, SERVO_MAX_US = 2000, SERVO_STEP_US = 10, SERVO_STEP_MS = 15;

// ---------------- LSM6DSO32 registers ----------------
const uint8_t LSM_WHO_AM_I = 0x0F, LSM_WHO_VALUE = 0x6C;
const uint8_t LSM_CTRL1_XL = 0x10, LSM_CTRL2_G = 0x11, LSM_CTRL3_C = 0x12;
const uint8_t LSM_STATUS = 0x1E, LSM_OUT_TEMP_L = 0x20;      // 0x20..0x2D: temp, gyro XYZ, accel XYZ
const float   LSM_ACC_MG_PER_LSB  = 0.488f;   // +-16 g
const float   LSM_GYR_MDPS_PER_LSB = 70.0f;   // +-2000 dps

// ---------------- BMP585 registers ----------------
const uint8_t BMP_CHIP_ID = 0x01, BMP_CHIP_VALUE = 0x51;
const uint8_t BMP_TEMP_XLSB = 0x1D, BMP_STATUS = 0x28;
const uint8_t BMP_OSR_CONFIG = 0x36, BMP_ODR_CONFIG = 0x37, BMP_CMD = 0x7E;

Servo servo;
bool servoRunning = true, lsmOk = false, bmpOk = false;
int servoUs = 1500, servoDir = +1;
uint32_t lastStep = 0, lastPrint = 0;

// ---------------- I2C helpers ----------------
bool writeReg(uint8_t addr, uint8_t reg, uint8_t val) {
  Wire.beginTransmission(addr);
  Wire.write(reg);
  Wire.write(val);
  return Wire.endTransmission() == 0;
}

bool readRegs(uint8_t addr, uint8_t reg, uint8_t *buf, uint8_t n) {
  Wire.beginTransmission(addr);
  Wire.write(reg);
  if (Wire.endTransmission(false) != 0) return false;          // repeated start
  if (Wire.requestFrom(addr, n) != n) return false;
  for (uint8_t i = 0; i < n; i++) buf[i] = Wire.read();
  return true;
}

void i2cScan() {
  Serial.println("I2C scan (Wire, SDA 18 / SCL 19):");
  int found = 0;
  for (uint8_t a = 1; a < 127; a++) {
    Wire.beginTransmission(a);
    if (Wire.endTransmission() == 0) {
      Serial.printf("  found 0x%02X%s\n", a, a == LSM_ADDR ? "  <- LSM6DSO32" : a == BMP_ADDR ? "  <- BMP585" : "");
      found++;
    }
  }
  if (!found) Serial.println("  nothing found -> check 3V3 on J8 pin 3, GND, SDA/SCL trails");
}

// ---------------- LSM6DSO32 ----------------
bool lsmBegin() {
  uint8_t who = 0;
  if (!readRegs(LSM_ADDR, LSM_WHO_AM_I, &who, 1)) { Serial.println("LSM6DSO32: no answer at 0x6A"); return false; }
  Serial.printf("LSM6DSO32 WHO_AM_I = 0x%02X (expect 0x6C)\n", who);
  if (who != LSM_WHO_VALUE) return false;
  writeReg(LSM_ADDR, LSM_CTRL3_C, 0x01);                       // software reset
  delay(20);
  bool ok = writeReg(LSM_ADDR, LSM_CTRL3_C, 0x44)              // BDU + register auto-increment
         && writeReg(LSM_ADDR, LSM_CTRL1_XL, 0x4C)             // accel 104 Hz, +-16 g
         && writeReg(LSM_ADDR, LSM_CTRL2_G, 0x4C);             // gyro  104 Hz, +-2000 dps
  delay(50);
  return ok;
}

bool lsmRead(int16_t &t, int16_t g[3], int16_t a[3]) {
  uint8_t b[14];
  if (!readRegs(LSM_ADDR, LSM_OUT_TEMP_L, b, 14)) return false;
  t = (int16_t)(b[1] << 8 | b[0]);
  for (int i = 0; i < 3; i++) {
    g[i] = (int16_t)(b[3 + 2 * i] << 8 | b[2 + 2 * i]);
    a[i] = (int16_t)(b[9 + 2 * i] << 8 | b[8 + 2 * i]);
  }
  return true;
}

// ---------------- BMP585 ----------------
bool bmpBegin() {
  uint8_t id = 0;
  writeReg(BMP_ADDR, BMP_CMD, 0xB6);                           // soft reset
  delay(10);
  if (!readRegs(BMP_ADDR, BMP_CHIP_ID, &id, 1)) { Serial.println("BMP585: no answer at 0x47"); return false; }
  Serial.printf("BMP585 CHIP_ID = 0x%02X (expect 0x51)\n", id);
  if (id != BMP_CHIP_VALUE) return false;
  uint8_t st = 0;
  readRegs(BMP_ADDR, BMP_STATUS, &st, 1);
  Serial.printf("BMP585 STATUS = 0x%02X (bit1 = NVM ready)\n", st);
  bool ok = writeReg(BMP_ADDR, BMP_ODR_CONFIG, 0x80 | (0x14 << 2) | 0x00)   // standby, 25 Hz, deep-standby off
         && writeReg(BMP_ADDR, BMP_OSR_CONFIG, 0x40 | (4 << 3) | 1);         // pressure on, OSR_P x16, OSR_T x2
  delay(5);
  ok = ok && writeReg(BMP_ADDR, BMP_ODR_CONFIG, 0x80 | (0x14 << 2) | 0x01); // normal mode
  delay(60);
  return ok;
}

bool bmpRead(int32_t &rawT, uint32_t &rawP) {
  uint8_t b[6];
  if (!readRegs(BMP_ADDR, BMP_TEMP_XLSB, b, 6)) return false;
  rawT = (int32_t)((uint32_t)b[2] << 16 | (uint32_t)b[1] << 8 | b[0]);
  if (rawT & 0x800000) rawT |= 0xFF000000;                     // sign-extend 24 bit
  rawP = (uint32_t)b[5] << 16 | (uint32_t)b[4] << 8 | b[3];
  return true;
}

// ---------------- servo ----------------
void attachServo(uint8_t pin) {
  servo.detach();
  servoPin = pin;
  servo.attach(servoPin, 500, 2500);
  servo.writeMicroseconds(servoUs);
  Serial.printf("servo on pin %u\n", servoPin);
}

void handleKeys() {
  while (Serial.available()) {
    char c = Serial.read();
    if (c == 's') { servoRunning = !servoRunning; Serial.println(servoRunning ? "servo: sweeping" : "servo: stopped"); }
    else if (c == 'c') { servoRunning = false; servoUs = 1500; servo.writeMicroseconds(servoUs); Serial.println("servo: centre 1500 us"); }
    else if (c >= '2' && c <= '5') attachServo(c - '0');
    else if (c == 'x') attachServo(10);
  }
}

void setup() {
  Serial.begin(115200);
  uint32_t t0 = millis();
  while (!Serial && millis() - t0 < 3000) {}
  Serial.println("\nVRITRA Board 4 test — Teensy 4.0");
  Wire.begin();
  Wire.setClock(100000);                                       // 100 kHz, same as the flight firmware
  i2cScan();
  lsmOk = lsmBegin();
  bmpOk = bmpBegin();
  Serial.printf("LSM6DSO32 %s, BMP585 %s\n", lsmOk ? "OK" : "FAIL", bmpOk ? "OK" : "FAIL");
  attachServo(servoPin);
  Serial.println("keys: s stop/start  c centre  2/3/4/5/x servo pin 2/3/4/5/10\n");
}

void loop() {
  handleKeys();
  uint32_t now = millis();

  if (servoRunning && now - lastStep >= SERVO_STEP_MS) {      // continuous sweep, never blocks the sensors
    lastStep = now;
    servoUs += servoDir * SERVO_STEP_US;
    if (servoUs >= SERVO_MAX_US) { servoUs = SERVO_MAX_US; servoDir = -1; }
    if (servoUs <= SERVO_MIN_US) { servoUs = SERVO_MIN_US; servoDir = +1; }
    servo.writeMicroseconds(servoUs);
  }

  if (now - lastPrint >= 100) {                                // 10 Hz print
    lastPrint = now;
    Serial.printf("t=%7lu ms  servo %u us | ", now, servoUs);
    int16_t t, g[3], a[3];
    if (lsmOk && lsmRead(t, g, a)) {
      Serial.printf("LSM raw A %6d %6d %6d  G %6d %6d %6d  T %6d | ", a[0], a[1], a[2], g[0], g[1], g[2], t);
      Serial.printf("A[g] %6.3f %6.3f %6.3f  G[dps] %7.2f %7.2f %7.2f  T %5.2f C | ",
                    a[0] * LSM_ACC_MG_PER_LSB / 1000.0f, a[1] * LSM_ACC_MG_PER_LSB / 1000.0f,
                    a[2] * LSM_ACC_MG_PER_LSB / 1000.0f, g[0] * LSM_GYR_MDPS_PER_LSB / 1000.0f,
                    g[1] * LSM_GYR_MDPS_PER_LSB / 1000.0f, g[2] * LSM_GYR_MDPS_PER_LSB / 1000.0f,
                    25.0f + t / 256.0f);
    } else {
      Serial.print("LSM -- | ");
    }
    int32_t rt; uint32_t rp;
    if (bmpOk && bmpRead(rt, rp)) {
      float tc = rt / 65536.0f, pa = rp / 64.0f;
      float alt = 44330.0f * (1.0f - powf(pa / 101325.0f, 0.1903f));
      Serial.printf("BMP raw T %ld P %lu  T %5.2f C  P %9.2f Pa  alt %7.2f m", (long)rt, (unsigned long)rp, tc, pa, alt);
    } else {
      Serial.print("BMP --");
    }
    Serial.println();
  }
}
```

**Power it like this, or the sensors won't answer:**
- **J8 (logic power):** needs **5 V on pin 1, GND on pins 2 and 4, and 3.3 V on pin 3**, from the power board or a bench supply. The LSM6DSO32 and BMP585 get their 3.3 V only from J8 pin 3, not from the Teensy. With USB alone, the Teensy runs but the I²C scan finds nothing.
- **J6 (servo power):** **5 V on pin 1 (+5VS), GND on pin 2** for the MG90S. Plug the servo into **J12** (brown −, red +, orange S) on Teensy pin 4.
- **USB and J8 at the same time:** the Teensy 4.0 joins USB 5 V and VIN unless you've cut the VIN–VUSB pad on its underside. So for this test, either cut that pad or briefly power from J8 only.

**What you should see:**
- I²C scan finds **0x47** (BMP585) and **0x6A** (LSM6DSO32).
- `WHO_AM_I = 0x6C` and `CHIP_ID = 0x51`, then `LSM6DSO32 OK, BMP585 OK`.
- Lying flat: accelerometer Z about **±1.0 g** (2049 raw), X/Y near 0, gyro near 0, both temperatures near room temperature.
- BMP pressure about **95 000–101 000 Pa**, depending on your altitude. Lift the board 1 m and the altitude rises by about 1 m.
- The servo sweeps 1000 → 2000 µs and back, about once every 3 seconds.

**CLAUDE** (2026-10-25T15:04:59)

Your test shows the servo and the Teensy pin both work, so the fault is in the resistor or its connection. A correct 330 Ω resistor in series doesn't stop a servo signal: the servo input draws almost no current, and 330 Ω drops only a few millivolts. So the part or a joint is wrong. Check in this order:

**1. Check the resistor's value**

| Value | 4-band colours | 5-band colours | Works? |
|---|---|---|---|
| **330 Ω** (correct) | orange orange **brown** gold | orange orange black **black** brown | yes |
| 3.3 kΩ | orange orange **red** | orange orange black **brown** | usually yes |
| **330 kΩ** (wrong) | orange orange **yellow** | orange orange black **orange** | **no**, signal too weak |

A 330 kΩ resistor in place of 330 Ω is the most common cause of exactly what you're seeing.

**2. Measure it on the board (power OFF), meter on Ω**
- Measure across the 2 legs of the resistor for that plug (J12/pin 4 = **R3**). It should read about **330 Ω**. A reading of hundreds of kΩ or open (OL) means a wrong or broken resistor.
- Measure from **Teensy pin 4** (the header pin) to **J12's S pin** (top pin). It should also read about **330 Ω**. Open here, with the resistor itself reading OK, means a cold solder joint or a gap in the trail between the Teensy, the resistor and J12.

**3. Measure with power ON, servo test running, meter on DC V**
- Teensy pin 4 to GND: about 0.2–0.4 V (the pulses averaged).
- J12 S pin to GND: about the same. 0 V here, with the Teensy side OK, means the path is broken.

**Fix:**
- **Wrong value:** fit a **330 Ω** (anything from 100 Ω to 1 kΩ works).
- **Bad joint:** reheat both resistor legs and the trail joints until the solder flows into the hole.
- **Can't find it quickly:** solder a wire link in place of the resistor. The servo works without it. The resistor only protects the Teensy pin if the servo lead shorts, so put a 330 Ω back later for flight.

The same check applies to every servo plug:

| Plug | Teensy pin | Resistor |
|---|---|---|
| J10 | 2 | R1 |
| J11 | 3 | R2 |
| J12 | 4 | R3 |
| J13 | 5 | R4 |
| J16 | 10 | R9 |
| J14 (ESC) | 6 | R5 |

If several plugs fail the same way, it's most likely a wrong-value batch, such as all 330 kΩ.

This program zeroes the attitude at boot (roll = pitch = yaw = 0) and then uses the gyro wheel to turn the CanSat back to its boot heading. It passes a syntax check with your Teensy compiler; it hasn't run on hardware.

**One wheel can only correct rotation around its own axis.** If the wheel shaft points along the can's long axis, the program stops the CanSat spinning or twisting under the parachute. It cannot actively level a tilt. The spinning wheel only resists tilting (gyroscopic stiffness), so "straight" here means "holds its boot heading".

**What I checked about your module:**
- LSM6DSO32 sensitivities used: ±2000 dps → 70 mdps/LSB, and ±16 g → 0.488 mg/LSB. The gyro figures are confirmed by published family values ([nRF Connect / Zephyr binding](https://nrfconnectdocs.nordicsemi.com/ncs/latest/zephyr/build/dts/api/bindings/sensor/st%2Clsm6dso32-i2c.html)). The accelerometer figure follows the binding's note that the LSM6DSO32 is double the LSM6DSO, which I couldn't check against ST's own table.
- Offsets: about **±1 dps gyro zero-rate** and about **±20 mg accelerometer zero-g**. I couldn't get the LSM6DSO32's own table; these come from the sister part LSM6DSOP ([datasheet table](https://www.alldatasheet.net/html-pdf/1211916/STMICROELECTRONICS/LSM6DSOP/40943/18/LSM6DSOP.html)), and ST's product page ([ST LSM6DSO32](https://st.com/en/mems-and-sensors/lsm6dso32.html)) has the full datasheet.
- These offsets differ on every chip, so fixed numbers from a datasheet are no use. The program measures your board's gyro bias at every boot (3 s, CanSat still) and stores that attitude as zero. If you move it during those 3 s, it rejects the reading and tries again.

```cpp
/*
 * VRITRA BOARD 4 — boot zeroing + gyro-wheel stabilisation (Teensy 4.0)
 *
 * WHAT IT DOES
 *  1. Boot: ESC gets 1000 us (stop) so it arms. Keep the CanSat STILL for ~3 s:
 *     the LSM6DSO32 gyro bias is measured and the present attitude is stored as
 *     roll = pitch = yaw = 0.
 *  2. Then roll / pitch / yaw are reported relative to that boot attitude.
 *  3. After you type 'm' (motor enable) the gyro wheel spins up to a base speed and
 *     a PD controller changes the wheel speed to push yaw (rotation about the wheel
 *     axis) back to 0: speeding the wheel up twists the CanSat one way, slowing it
 *     down twists it the other way (reaction torque).
 *
 * PHYSICS LIMIT: one wheel can only correct rotation AROUND ITS OWN AXIS (WHEEL_AXIS).
 * Tilt about the other two axes is not corrected actively; the spinning wheel only
 * resists it (gyroscopic stiffness).
 *
 * CONNECTIONS (Board 4)
 *  LSM6DSO32 0x6A on pins 18 SDA / 19 SCL (3.3 V from J8 pin 3)
 *  ESC signal  pin 6 -> R5 330 R -> J14 S  (Cyclone 45A BLHeli_S, normal one-direction mode)
 *  Hall sensor pin 9 <- J15 OUT (10k pull-up)  -> wheel RPM (display only)
 *
 * KEYS: m = motor enable / disable    z = zero again (CanSat must be still)
 *       x = STOP motor now            + / - = raise / lower wheel base speed
 *       r = reverse correction sign (if the wheel makes the spin WORSE)
 *
 * FIRST TEST: wheel secured, CanSat on a turntable or hanging on a string, props/
 * parachute off, people clear. The motor only starts after 'm'.
 */
#include <Wire.h>
#include <Servo.h>

// ---------- settings ----------
const uint8_t LSM_ADDR   = 0x6A;      // LSM6DSO32 (DO -> GND)
const uint8_t ESC_PIN    = 6;         // J14 S via R5
const uint8_t HALL_PIN   = 9;         // J15 OUT
const uint8_t WHEEL_AXIS = 2;         // LSM axis parallel to the wheel shaft: 0 = X, 1 = Y, 2 = Z
const uint8_t MAGNETS    = 1;         // Hall pulses per wheel turn

const int ESC_STOP_US = 1000;         // BLHeli_S stop / arm
const int ESC_MIN_US  = 1150;         // lowest running throttle (wheel never stops while enabled)
const int ESC_MAX_US  = 1700;         // safety ceiling for tests (raise only after testing)
int       escBaseUs   = 1400;         // base wheel speed, change with + / -

float KP = 4.0f;                      // us per degree of yaw error
float KD = 1.5f;                      // us per deg/s of yaw rate
int   corrSign = +1;                  // flip with 'r' if correction acts the wrong way
const float TILT_CUTOFF_DEG = 70.0f;  // motor stops if the CanSat tips over this far

// LSM6DSO32 scaling (ST datasheet): +-16 g = 0.488 mg/LSB, +-2000 dps = 70 mdps/LSB
const float ACC_G_PER_LSB    = 0.000488f;
const float GYR_DPS_PER_LSB  = 0.070f;
const float LOOP_HZ          = 200.0f;
const float COMP_ALPHA       = 0.98f; // complementary filter: 98 % gyro, 2 % accelerometer

Servo esc;

// ---------- state ----------
float gyroBias[3] = {0, 0, 0};        // deg/s, measured at boot
float roll0 = 0, pitch0 = 0;          // boot attitude from the accelerometer
float roll = 0, pitch = 0, yaw = 0;   // attitude relative to boot, degrees
float rate[3] = {0, 0, 0};            // bias-corrected rates, deg/s
bool  motorOn = false, sensorOk = false;
int   escUs = ESC_STOP_US;
volatile uint32_t hallPulses = 0;
void hallIsr() { hallPulses++; }

// ---------- I2C ----------
bool writeReg(uint8_t r, uint8_t v) {
  Wire.beginTransmission(LSM_ADDR); Wire.write(r); Wire.write(v);
  return Wire.endTransmission() == 0;
}
bool readRegs(uint8_t r, uint8_t *buf, uint8_t n) {
  Wire.beginTransmission(LSM_ADDR); Wire.write(r);
  if (Wire.endTransmission(false) != 0) return false;
  if (Wire.requestFrom(LSM_ADDR, n) != n) return false;
  for (uint8_t i = 0; i < n; i++) buf[i] = Wire.read();
  return true;
}

// ---------- LSM6DSO32 ----------
bool lsmStart() {
  uint8_t who = 0;
  if (!readRegs(0x0F, &who, 1) || who != 0x6C) { Serial.printf("LSM6DSO32 not found (0x%02X)\n", who); return false; }
  writeReg(0x12, 0x01); delay(20);    // software reset
  writeReg(0x12, 0x44);               // BDU (consistent 16-bit reads) + address auto-increment
  writeReg(0x10, 0x6C);               // accel 416 Hz, +-16 g
  writeReg(0x11, 0x6C);               // gyro  416 Hz, +-2000 dps
  delay(50);
  return true;
}

// read accel (g) and gyro (deg/s, raw, bias NOT removed)
bool lsmRead(float acc[3], float gyr[3]) {
  uint8_t b[12];
  if (!readRegs(0x22, b, 12)) return false;   // 0x22..0x27 gyro XYZ, 0x28..0x2D accel XYZ
  for (int i = 0; i < 3; i++) {
    gyr[i] = (int16_t)(b[1 + 2 * i] << 8 | b[2 * i]) * GYR_DPS_PER_LSB;
    acc[i] = (int16_t)(b[7 + 2 * i] << 8 | b[6 + 2 * i]) * ACC_G_PER_LSB;
  }
  return true;
}

float accRoll(const float a[3])  { return atan2f(a[1], a[2]) * 57.2958f; }
float accPitch(const float a[3]) { return atan2f(-a[0], sqrtf(a[1] * a[1] + a[2] * a[2])) * 57.2958f; }

// ---------- boot zeroing: CanSat must be still ----------
bool calibrate() {
  Serial.println("ZEROING: keep the CanSat still ...");
  const int N = 600;                  // 600 samples at ~200 Hz = 3 s
  double gs[3] = {0, 0, 0}, gsq[3] = {0, 0, 0}, as[3] = {0, 0, 0};
  for (int k = 0; k < N; k++) {
    float a[3], g[3];
    if (!lsmRead(a, g)) { Serial.println("read error during zeroing"); return false; }
    for (int i = 0; i < 3; i++) { gs[i] += g[i]; gsq[i] += g[i] * g[i]; as[i] += a[i]; }
    delay(5);
  }
  float a[3];
  for (int i = 0; i < 3; i++) {
    gyroBias[i] = gs[i] / N;
    float var = gsq[i] / N - gyroBias[i] * gyroBias[i];
    a[i] = as[i] / N;
    if (var > 0.5f * 0.5f) {          // gyro moved more than ~0.5 deg/s rms -> not still
      Serial.printf("moved during zeroing (axis %d noise %.2f dps) -> try again\n", i, sqrtf(var));
      return false;
    }
  }

    else if (c == 'z') { motorStop("re-zero"); while (!calibrate()) delay(1000); }
    else if (c == '+') { escBaseUs = min(escBaseUs + 25, ESC_MAX_US - 100); Serial.printf("base %d us\n", escBaseUs); }
    else if (c == '-') { escBaseUs = max(escBaseUs - 25, ESC_MIN_US + 100); Serial.printf("base %d us\n", escBaseUs); }
    else if (c == 'r') { corrSign = -corrSign; Serial.printf("correction sign %+d\n", corrSign); }
  }

  uint32_t nowUs = micros();
  if (nowUs - tLoop < (uint32_t)(1e6f / LOOP_HZ)) return;
  float dt = (nowUs - tLoop) * 1e-6f;
  tLoop = nowUs;

    if (motorOn) motorStop("sensor read failed");
    return;

  float accR = accRoll(a) - roll0, accP = accPitch(a) - pitch0;
  roll  = COMP_ALPHA * (roll  + rate[0] * dt) + (1 - COMP_ALPHA) * accR;
  pitch = COMP_ALPHA * (pitch + rate[1] * dt) + (1 - COMP_ALPHA) * accP;
  yaw  += rate[WHEEL_AXIS] * dt;                  // yaw = rotation about the wheel axis (no absolute reference)
  if (yaw > 180) yaw -= 360;
  if (yaw < -180) yaw += 360;

  // ----- wheel control -----
  if (motorOn) {
    if (fabsf(roll) > TILT_CUTOFF_DEG || fabsf(pitch) > TILT_CUTOFF_DEG) {
      motorStop("tipped over");
    } else {
      float yawErr = yaw;                          // target = 0 (boot heading)
      float u = corrSign * (KP * yawErr + KD * rate[WHEEL_AXIS]);
      int target = constrain((int)(escBaseUs + u), ESC_MIN_US, ESC_MAX_US);
      // limit how fast the throttle may change (protects ESC, battery, structure)
      int stepMax = 4;                             // us per loop = 800 us/s
      int next = escUs < ESC_MIN_US ? ESC_MIN_US : escUs;
      next += constrain(target - next, -stepMax, stepMax);
      setEsc(next);
      yawErrPrev = yawErr;
    }
  }

  // ----- wheel RPM from the Hall sensor (once per second) -----
  if (millis() - tRpm >= 1000) {
    uint32_t p = hallPulses;
    rpm = (p - hallLast) * 60UL / MAGNETS;
    hallLast = p; tRpm = millis();
  }

  // ----- report 10 Hz -----
  if (millis() - tPrint >= 100) {
    tPrint = millis();
    Serial.printf("roll %7.2f  pitch %7.2f  yaw %7.2f deg | rate %7.2f %7.2f %7.2f dps | ESC %4d us %s | wheel %5lu rpm\n",
                  roll, pitch, yaw, rate[0], rate[1], rate[2], escUs, motorOn ? "ON " : "off", (unsigned long)rpm);
  }
  (void)yawErrPrev;
}
```

```cpp
  Serial.printf("[4] DS3231 OK  20%02u-%02u-%02u %02u:%02u:%02u\n", bcd(t[6]), bcd(t[5] & 0x1F), bcd(t[4]),
                bcd(t[2] & 0x3F), bcd(t[1]), bcd(t[0] & 0x7F));

void spiTests() {

  SPI1.beginTransaction(s);
  digitalWrite(FLASH_CS, LOW);
  SPI1.transfer(0x9F);
  uint8_t m = SPI1.transfer(0), t = SPI1.transfer(0), c = SPI1.transfer(0);
  digitalWrite(FLASH_CS, HIGH);
  SPI1.endTransaction();
  Serial.printf("[5] W25Q64 JEDEC %02X %02X %02X %s\n", m, t, c, (m == 0xEF && c == 0x17) ? "OK" : "FAIL (want EF 40 17)");

  SPI1.beginTransaction(s);
  digitalWrite(LORA_CS, LOW);
```

## 3. INA260 wiring (you have one)
Put it in the **battery position (U1)** on the POWER board, in series with the main battery line:
- Battery **+** → rocker switch → fuse → INA260 **IN+**.
- INA260 **IN−** → the VBAT bus, which feeds the ESC and the buck converters.
- **VCC** → 3.3 V, **GND** → GND, **SDA/SCL** → POWER J2 → Board 1 J2 → Teensy pins 18/19.
- Leave the address pins at GND (address 0x40). Address 0x47 would clash with the BMP585.
- Bridge the two empty INA positions (U2, U3) with plain wire from input to output.

**Charging status:** a positive current means the pack is discharging. A negative current only appears if the charger connects on the IN− side. The INA260 can't see a charger clipped straight onto the battery leads.

**Before wiring:** I couldn't find a 7Semi pinout for this board. Check on yours whether its VBUS pin is tied to IN+ or IN−.

## 4. INA280
The INA280 is an **analog** current-sense amplifier, not an I²C sensor:
- You need your own shunt resistor, and its output goes to an analog pin.
- It works with input voltages from 2.7 V to 120 V and gains of 20 to 500 V/V (by part number).
- It doesn't measure voltage or power itself.

The INA260 you already have is the better fit here.
