from pathlib import Path
import json, os, subprocess, sys

root=Path(__file__).resolve().parent.parent
os.chdir(root)
build=root/'build';build.mkdir(exist_ok=True)
checks=[]
def run(args):
    result=subprocess.run(args,cwd=root,check=True,text=True,capture_output=True)
    output=(result.stdout+result.stderr).strip()
    print(output or 'PASS: '+' '.join(str(a) for a in args))
    checks.append({'command':[str(a) for a in args],'exit_code':result.returncode,'output':output})
for name in ('test_core','test_modules'):
    exe=build/(name+('.exe' if os.name=='nt' else ''))
    run(['g++','-std=c++17','-Wall','-Wextra','-Werror','-pedantic',
         f'deliverables/reference_code/{name}.cpp','-o',str(exe)])
    run([str(exe)])
run([sys.executable,'deliverables/reference_code/test_gcs_protocol.py'])
run([sys.executable,'deliverables/reference_code/engineering_checks.py'])
run([sys.executable,'research/complete_handoff.py'])
Path('research/check_results.json').write_text(json.dumps(checks,indent=2)+'\n',encoding='utf-8')
